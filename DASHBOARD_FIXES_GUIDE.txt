19. Cache Query Results (For Better Performance)

    ' Add class-level variables
    Private _cacheTime As DateTime
    Private _cachedData As DataTable
    
    Private Function GetCachedData(query As String, ParamArray params()) As DataTable
        ' Refresh cache every 5 minutes
        If _cachedData Is Nothing OrElse 
           DateTime.Now.Subtract(_cacheTime).TotalMinutes > 5 Then
            _cachedData = GetDataTable(query, params)
            _cacheTime = DateTime.Now
        End If
        Return _cachedData
    End Function
