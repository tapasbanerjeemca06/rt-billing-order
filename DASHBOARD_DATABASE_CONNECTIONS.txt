VIEW 1: SALES ANALYSIS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Query: SELECT TOP 10 i.ItemName, SUM(bd.Amount) AS Total 
       FROM tblBillDetail bd 
       INNER JOIN tblItem i ON bd.ItemID=i.ItemID 
       INNER JOIN tblBill b ON bd.BillID=b.BillID 
       WHERE b.BillDate BETWEEN @p0 AND @p1 
       GROUP BY i.ItemName 
       ORDER BY Total DESC

Tables Used:
- tblBillDetail (bd) - Bill line items
- tblItem (i) - Item master
- tblBill (b) - Bill header

Chart: Bar & Pie showing top 10 products by revenue
Table Columns: ItemName, Total

VIEW 2: EXPENSE BREAKDOWN  
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Query: SELECT ExpHead AS Category, SUM(Amount) AS Total 
       FROM tblExpense 
       WHERE ExpDate BETWEEN @p0 AND @p1 
       GROUP BY ExpHead 
       ORDER BY Total DESC

Tables Used:
- tblExpense - Expense entries

Chart: Bar & Pie showing expenses by category
Table Columns: Category (ExpHead), Total

VIEW 3: AGENT PERFORMANCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Query: SELECT a.AgentName, COUNT(b.BillID) AS Bills, 
              ISNULL(SUM(b.GrandTotal),0) AS Total 
       FROM tblAgent a 
       LEFT JOIN tblBill b ON a.AgentID=b.AgentID 
           AND b.BillDate BETWEEN @p0 AND @p1 
       GROUP BY a.AgentName 
       HAVING COUNT(b.BillID) > 0 
       ORDER BY Total DESC

Tables Used:
- tblAgent (a) - Agent master
- tblBill (b) - Bills

Chart: Bar & Pie showing agent sales comparison
Table Columns: AgentName, Bills (count), Total (revenue)

VIEW 4: TOP PRODUCTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Query: SELECT TOP 10 i.ItemName, SUM(bd.Qty) AS QtySold, 
              SUM(bd.Amount) AS Revenue 
       FROM tblBillDetail bd 
       INNER JOIN tblItem i ON bd.ItemID=i.ItemID 
       INNER JOIN tblBill b ON bd.BillID=b.BillID 
       WHERE b.BillDate BETWEEN @p0 AND @p1 
       GROUP BY i.ItemName 
       ORDER BY Revenue DESC

Tables Used:
- tblBillDetail (bd) - Bill line items
- tblItem (i) - Item master  
- tblBill (b) - Bill header

Chart: Bar & Pie showing top products by quantity & revenue
Table Columns: ItemName, QtySold, Revenue
